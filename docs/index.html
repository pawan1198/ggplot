example.R — build a complete ggplot2 visualization 
# Load the library
library(ggplot2)

# Create a polished scatter plot with smooth line
ggplot(mtcars, aes(x = wt, y = mpg, color = factor(cyl))) +
  geom_point(size = 3, alpha = 0.8) +
  geom_smooth(method = "lm", se = FALSE, lwd = 0.8) +
  scale_color_manual(values = c("#e84855", "#2d6be4", "#12a36a")) +
  labs(
    title   = "Car Weight vs. Fuel Efficiency",
    subtitle= "Grouped by number of cylinders",
    caption = "Source: Motor Trend US magazine, 1974",
    x       = "Weight (1000 lbs)",
    y       = "Miles per Gallon",
    color   = "Cylinders"
  ) +
  theme_minimal(base_size = 13) +
  theme(legend.position = "top",
        plot.title = element_text(face = "bold"))
