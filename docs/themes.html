Themes & Styling

Built-in themes, ggthemes, hrbrthemes, ggdark, mdthemes, ggthemr, ggtech — with full customisation

A complete guide to ggplot2 themes — all 8 built-in themes plus extension packages ggthemes, hrbrthemes, ggdark, mdthemes, ggthemr and ggtech. Includes theme_set() and full theme() customisation.

Built-in themes

ggplot2 ships with 8 themes. The default is theme_grey() — you never need to specify it unless you want to be explicit.

install.packages("ggplot2")
library(ggplot2)
Function Description
theme_grey() Default — grey background, white grid lines
theme_bw() Black & white — white panel, dark border
theme_linedraw() Black lines on white background only
theme_light() Like linedraw but with grey lines and axes
theme_dark() Dark panel background
theme_minimal() No background annotations — very clean
theme_classic() No grid lines — axes only
theme_void() Completely empty — blank canvas

Examples

library(ggplot2)
iris -> dataset

p <- ggplot(dataset, aes(x = Sepal.Length, y = Sepal.Width, fill = Species)) +
  geom_bar(stat = "identity", position = "stack") +
  facet_wrap(~ Species)

p + theme_grey()      # default
p + theme_bw(base_size = 14)
p + theme_minimal()
p + theme_classic()
p + theme_dark()
p + theme_light()
p + theme_linedraw()
p + theme_void()
Tipbase_size argument

All built-in themes accept a base_size argument to scale all text proportionally:

p + theme_bw(base_size = 20)   # larger text
p + theme_minimal(base_size = 9) # smaller text

ggdark

ggdark provides dark mode versions of all 8 built-in ggplot2 themes. It can also convert any theme into dark mode with dark_mode().

install.packages("ggdark")
library(ggplot2)
library(ggdark)

Dark theme functions

dark_theme_gray()       dark_theme_bw()
dark_theme_linedraw()   dark_theme_light()
dark_theme_dark()       dark_theme_minimal()
dark_theme_classic()    dark_theme_void()

# Convert any theme to dark mode
dark_mode(theme_economist())
dark_mode(theme_ipsum())

Usage examples

library(ggplot2)
library(ggdark)

mtcars -> dt
cyls <- as.factor(dt$cyl)

p <- ggplot(mtcars, aes(x = mpg, fill = cyls)) +
  geom_density(alpha = 0.7) +
  theme(legend.position = "top")

p + dark_theme_classic()
p + dark_theme_linedraw()
p + dark_theme_minimal()

# Dark mode on any theme
p + dark_mode(theme_minimal())
p + dark_mode(theme_economist())

mdthemes

mdthemes wraps all themes from ggplot2, ggthemes, hrbrthemes and other packages with Markdown support — plot titles and labels can use **bold** and *italic* syntax.

install.packages("mdthemes")
library(ggplot2)
library(mdthemes)
library(ggplot2)
library(mdthemes)

cyls <- as.factor(mtcars$cyl)

p <- ggplot(mtcars, aes(x = mpg, fill = cyls)) +
  geom_density(alpha = 0.7) +
  labs(title    = "**Fuel Economy** by *Cylinders*",
       subtitle = "**Bold** and *italic* in titles") +
  theme(legend.position = "top")

# Regular theme — markdown NOT rendered
p + theme_bw()

# md version — markdown IS rendered
p + md_theme_bw()
p + md_theme_minimal()
p + md_theme_classic()

ggthemr

ggthemr works differently — it sets a global theme for all subsequent plots using ggthemr(), rather than being added as a layer.

# Install from GitHub
# devtools::install_github('Mikata-Project/ggthemr')
library(ggthemr)
library(ggplot2)
# Set a global theme
ggthemr("flat")

cyls <- as.factor(mtcars$cyl)
ggplot(mtcars, aes(x = mpg, fill = cyls)) +
  geom_density(alpha = 0.7) +
  theme(legend.position = "top")

# ALWAYS reset after use
ggthemr_reset()

Available palettes

ggthemr("camouflage")     ggthemr("carrot")
ggthemr("chalk")          ggthemr("copper")
ggthemr("dust")           ggthemr("earth")
ggthemr("fresh")          ggthemr("grape")
ggthemr("grass")          ggthemr("greyscale")
ggthemr("light")          ggthemr("lilac")
ggthemr("pale")           ggthemr("sea")

WarningAlways reset ggthemr

ggthemr sets a global state. Always call ggthemr_reset() when done to restore the default ggplot2 theme.

ggtech

ggtech provides themes inspired by major tech companies.

# Install from GitHub
# devtools::install_github("ricardo-bion/ggtech", dependencies = TRUE)
library(ggtech)
library(ggplot2)
cyls <- as.factor(mtcars$cyl)

ggplot(mtcars, aes(x = mpg, fill = cyls)) +
  geom_density(alpha = 0.7) +
  ggtitle("Tech company themes") +
  theme_tech(theme = "airbnb") +
  scale_fill_tech(theme = "airbnb") +
  theme(legend.position = "top")

theme_tech(theme = "etsy")      # Etsy orange
theme_tech(theme = "X23andme")  # 23andMe

Setting a global theme

Use theme_set() to apply a theme to all subsequent plots in your session:

library(ggplot2)

# Save current theme and set a new one
old <- theme_set(theme_minimal(base_size = 13))

# Now ALL plots use theme_minimal
ggplot(mtcars, aes(x = wt, y = mpg)) + geom_point()
ggplot(iris, aes(x = Sepal.Width)) + geom_histogram()

# Restore the previous theme
theme_set(old)

Custom theme with theme()

Build your own reusable theme:

library(ggplot2)

my_theme <- theme_minimal(base_size = 13) +
  theme(
    # Titles
    plot.title      = element_text(face = "bold", size = 16,
                                   colour = "#0e0f14", margin = margin(b = 8)),
    plot.subtitle   = element_text(colour = "#6b6760", size = 12,
                                   margin = margin(b = 12)),
    plot.caption    = element_text(colour = "#aaaaaa", size = 9, hjust = 1),

    # Axes
    axis.title      = element_text(size = 11, colour = "#3d3e47"),
    axis.text       = element_text(size = 10, colour = "#6b6760"),
    axis.line       = element_line(colour = "#d8d2c8"),

    # Grid
    panel.grid.major = element_line(colour = "#ede8df", linewidth = 0.4),
    panel.grid.minor = element_blank(),
    panel.background = element_rect(fill = "#faf8f4", colour = NA),
    plot.background  = element_rect(fill = "#ffffff", colour = NA),

    # Legend
    legend.position  = "top",
    legend.title     = element_text(face = "bold", size = 10),
    legend.text      = element_text(size = 10),

    # Facet strips
    strip.background = element_blank(),
    strip.text       = element_text(face = "bold", size = 11),

    # Margins
    plot.margin      = margin(16, 16, 16, 16)
  )

# Apply to any plot
ggplot(mtcars, aes(x = wt, y = mpg, colour = factor(cyl))) +
  geom_point(size = 3, alpha = 0.85) +
  scale_colour_manual(values = c("#e84855","#2d6be4","#12a36a")) +
  labs(title = "Custom Theme", colour = "Cylinders") +
  my_theme
