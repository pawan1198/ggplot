Quick Reference

All key ggplot2 functions, theme arguments, scales, and packages — at a glance

Complete ggplot2 quick reference — geoms, aesthetics, scales, coordinates, facets, themes, combining plots, and full theme() argument table.

Essential packages

pkgs <- c(
  "ggplot2",     # Core visualization
  "ggthemes",    # Extra themes (economist, wsj, fivethirtyeight…)
  "hrbrthemes",  # Typography-centric themes
  "ggdark",      # Dark mode themes
  "mdthemes",    # Markdown-enabled themes
  "patchwork",   # Combine plots with operators
  "ggrepel",     # Non-overlapping labels
  "ggtext",      # Markdown/HTML in text elements
  "gridExtra",   # grid.arrange() layouts
  "cowplot",     # plot_grid() publication layouts
  "scales",      # Axis formatting helpers
  "dplyr",       # Data wrangling
  "tidyr"        # Reshaping data
)
install.packages(pkgs)

ggplot2 template

library(ggplot2)

ggplot(data = <DATA>, mapping = aes(<MAPPINGS>)) +
  <GEOM_FUNCTION>(
    mapping  = aes(<MAPPINGS>),
    stat     = <STAT>,
    position = <POSITION>   # "identity","stack","dodge","fill","jitter"
  ) +
  <SCALE_FUNCTION>   +
  <COORDINATE_FUNCTION> +
  <FACET_FUNCTION> +
  <THEME_FUNCTION>

Aesthetics

Aesthetic Maps to Use
x, y Axis position All geoms
colour Stroke / point colour Points, lines, box outlines

size Point size / line width Points, lines
shape Point shape (0–24) Points
alpha Transparency (0–1) All geoms
linetype Line style (1–6) Lines, borders
label Text string geom_text, geom_label
group Grouping (no visual) Lines, polygons

geom_tile() Heatmap tiles fill

theme() — key arguments

theme(
  # Text
  plot.title       = element_text(face="bold", size=16, hjust=0),
  plot.subtitle    = element_text(colour="grey50"),
  plot.caption     = element_text(hjust=1, size=9),
  axis.title       = element_text(size=11),
  axis.text        = element_text(size=10),
  axis.text.x      = element_text(angle=45, hjust=1),
  legend.title     = element_text(face="bold"),
  strip.text       = element_text(face="bold"),

  # Rectangles
  plot.background  = element_rect(fill="white"),
  panel.background = element_rect(fill="#f5f5f5"),
  panel.border     = element_rect(fill="transparent", colour="grey80"),
  legend.background= element_rect(fill="white"),
  strip.background = element_blank(),

  # Lines
  panel.grid.major = element_line(colour="grey90"),
  panel.grid.minor = element_blank(),
  axis.line        = element_line(colour="grey30"),
  axis.ticks       = element_blank(),

  # Positioning
  legend.position  = "top",   # "top","bottom","left","right","none",c(x,y)
  legend.direction = "horizontal",
  legend.key.size  = unit(0.8, "cm"),
  plot.margin      = margin(16, 16, 16, 16),
  panel.spacing    = unit(1, "lines"),
  plot.title.position = "plot"
)

Combining plots

library(patchwork)
p1 + p2                    # side by side
p1 / p2                    # stacked
(p1 | p2) / p3             # nested
p1 + p2 + plot_layout(ncol = 2)
p1 + p2 & theme_minimal()  # apply to all
p1 + p2 + plot_annotation(title = "Title", tag_levels = "A")
wrap_plots(list_of_plots, ncol = 2)
plot_spacer()               # empty gap

library(gridExtra)
grid.arrange(p1, p2, ncol = 2)
grid.arrange(p1, p2, p3, layout_matrix = matrix(c(1,1,2,3), ncol=2, byrow=TRUE))

library(cowplot)
plot_grid(p1, p2, labels = "AUTO", align = "hv", rel_widths = c(2,1))

Common patterns

Complete styled scatter

ggplot(mtcars, aes(x = wt, y = mpg, colour = factor(cyl))) +
  geom_point(size = 3, alpha = 0.85) +
  geom_smooth(method = "lm", se = FALSE, linewidth = 0.8) +
  scale_colour_manual(values = c("4"="#e84855","6"="#2d6be4","8"="#12a36a"),
                      name = "Cylinders") +
  labs(title    = "Vehicle Weight vs Fuel Economy",
       subtitle = "mtcars dataset — 1974 Motor Trend",
       caption  = "Source: Henderson & Velleman (1981)",
       x = "Weight (1,000 lbs)", y = "Miles per Gallon") +
  theme_minimal(base_size = 13) +
  theme(legend.position  = "top",
        plot.title       = element_text(face = "bold"),
        panel.grid.minor = element_blank())

Reusable custom theme

my_theme <- theme_minimal(base_size = 13) +
  theme(
    plot.title       = element_text(face = "bold", size = 15),
    plot.subtitle    = element_text(colour = "#6b6760", margin = margin(b=10)),
    panel.grid.minor = element_blank(),
    panel.grid.major = element_line(colour = "#ede8df"),
    legend.position  = "top",
    plot.margin      = margin(16, 16, 16, 16)
  )

ggplot(mtcars, aes(x = wt, y = mpg)) +
  geom_point() + labs(title = "My plot") + my_theme

Faceted with highlight

ggplot(mtcars, aes(x = wt, y = mpg, colour = factor(cyl))) +
  geom_point(data = transform(mtcars, cyl = NULL), colour = "grey85", size=2) +
  geom_point(size = 2.5) +
  facet_wrap(~ factor(cyl), labeller = label_both) +
  scale_colour_manual(values = c("4"="#e84855","6"="#2d6be4","8"="#12a36a")) +
  theme_minimal() +
  theme(legend.position = "none", strip.text = element_text(face = "bold"))
