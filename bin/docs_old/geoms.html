Module 08 — Visual Elements

Geom Function Gallery

Geoms are the geometric shapes used to represent data. Here is a complete reference with working code for every major geom in ggplot2.

Scatter Plots — geom_point()

R geom_point 
library(ggplot2)

# Basic scatter
ggplot(mtcars, aes(x = mpg, y = wt)) +
  geom_point()

# Styled scatter with regression line
ggplot(mtcars, aes(x = wt, y = mpg)) +
  geom_point(size = 2, color = "blue") +
  geom_smooth(method = "lm", se = FALSE)

# Color-mapped scatter with 3 variables
ggplot(mtcars, aes(x = mpg, y = wt, color = factor(cyl))) +
  geom_point(size = 3, alpha = 0.8) +
  scale_color_manual(values = c("#e84855","#2d6be4","#12a36a")) +
  labs(color = "Cylinders")

Line Charts — geom_line()

R geom_line 
# Basic line chart
ggplot(economics, aes(x = date, y = unemploy)) +
  geom_line()

# Multiple lines
ggplot(economics_long, aes(x = date, y = value01, color = variable)) +
  geom_line()

# geom_path — connects points in order they appear in data
ggplot(economics, aes(x = unemploy/pop, y = psavert)) +
  geom_path(alpha = 0.5)

Bars & Histograms

R geom_bar & geom_histogram 
# geom_bar — count by category (stat="count" is default)
ggplot(mtcars, aes(x = factor(cyl))) +
  geom_bar()

# geom_bar — pre-summarized (stat="identity")
ggplot(df1, aes(x = group, y = values)) +
  geom_bar(stat = "identity")

# geom_col — shorthand for stat="identity"
ggplot(df1, aes(x = group, y = values)) +
  geom_col()

# geom_histogram — continuous data
ggplot(mtcars, aes(x = mpg)) +
  geom_histogram(bins = 10, fill = "steelblue", color = "white")

# geom_histogram with binwidth
ggplot(mtcars, aes(x = mpg)) +
  geom_histogram(binwidth = 2, fill = "steelblue")

Distribution — geom_boxplot() & geom_violin()

R Boxplot and violin 
# Box plot
ggplot(mtcars, aes(x = factor(cyl), y = mpg)) +
  geom_boxplot(fill = "steelblue")

# Violin plot
ggplot(mtcars, aes(x = factor(cyl), y = mpg)) +
  geom_violin(fill = "steelblue", alpha = 0.6)

# Combined violin + box
ggplot(mtcars, aes(x = factor(cyl), y = mpg)) +
  geom_violin(fill = "lightblue") +
  geom_boxplot(width = 0.1)   # Narrow box overlaid

Density — geom_density()

R Density plots 
# Basic density plot
ggplot(mtcars, aes(x = mpg)) +
  geom_density(fill = "steelblue", alpha = 0.5)

# Multiple overlapping densities
cyls <- as.factor(mtcars$cyl)
ggplot(mtcars, aes(x = mpg, fill = cyls)) +
  geom_density(alpha = 0.7) +
  theme(legend.position = "top")

Area — geom_area()

R Area plots 
# Basic area plot
ggplot(economics, aes(x = date, y = unemploy)) +
  geom_area(fill = rgb(0, 0.5, 1, alpha = 0.5))

# Ribbon — upper and lower bounds
ggplot(economics, aes(x = date)) +
  geom_ribbon(aes(ymin = pce - 200, ymax = pce + 200),
              fill = "steelblue", alpha = 0.4)

Heatmap — geom_tile()

R Heatmap with geom_tile 
# Create correlation matrix heatmap
cor_data <- as.data.frame(cor(mtcars))
cor_data$var1 <- rownames(cor_data)

library(tidyr)
cor_long <- pivot_longer(cor_data, -var1, names_to="var2", values_to="corr")

ggplot(cor_long, aes(x=var1, y=var2, fill=corr)) +
  geom_tile() +
  scale_fill_gradient2(low="#2d6be4", mid="white", high="#e84855") +
  theme_minimal() +
  theme(axis.text.x = element_text(angle=45, hjust=1))
