Module 06 — Layout

Combining Multiple Plots

ggplot2 doesn't include built-in multi-plot layout, but three excellent packages fill this gap: patchwork, gridExtra, and cowplot.

Create Sample Plots First

Let's create four plots (p1–p4) that we'll combine throughout this module.

R Create p1, p2, p3, p4 
library(ggplot2)

mtcars -> df

# p1: Box plot
p1 <- ggplot(df, aes(x = "", y = mpg)) +
  geom_boxplot()

# p2: Normal density curve
p2 <- ggplot() +
  stat_function(fun = dnorm, geom = "density",
                xlim = c(-3, 3), fill = "white")

# p3: Line chart
p3 <- ggplot(df, aes(x = mpg, y = disp)) +
  geom_line(color = "gray20")

# p4: Scatter plot
p4 <- ggplot(df, aes(x = hp, y = wt)) +
  geom_point(color = "gray20")

patchwork — Recommended

patchwork is the most intuitive package for combining ggplot2 plots. It uses arithmetic operators to specify layout.

⭐
Recommendation: patchwork is the most modern and readable approach. If you're starting fresh, use patchwork.

Basic operators

R patchwork operators 
install.packages("patchwork")
library(patchwork)

p1 + p2              # Side by side (2 columns)
p1 | p2              # Same as + (columns)
p1 / p2              # Stack (rows)
p1 + p2 + p3 + p4    # 4 plots in a 2×2 grid (auto)
(p1 | p2) / p3       # Two on top, one on bottom

Controlling layout with plot_layout()

R plot_layout() — sizes and columns 
# Control number of columns
p1 + p2 + p3 + p4 +
  plot_layout(ncol = 3)

# Relative widths and heights
p1 + p2 + p3 + p4 +
  plot_layout(widths = c(2, 1), heights = c(1, 2))

# Custom design using letter codes
# 1=p1, 2=p2, 3=p3, 4=p4, #=empty space
p1 + p2 + p3 + p4 +
  plot_layout(design = "111
                         2#3
                         443")

wrap_plots() — for variable numbers of plots

R wrap_plots() 
# Useful when you don't know the number of plots at coding time
wrap_plots(p1, p2, p3, p4,
           ncol    = 2,
           nrow    = 2,
           widths  = c(1, 0.5),
           heights = c(0.5, 1))

# From a list of plots (e.g. from lapply)
plot_list <- list(p1, p2, p3, p4)
wrap_plots(plot_list, ncol = 2)

Adding spacers

R plot_spacer() 
# Add blank spaces between plots
p1 + plot_spacer() + p2      # p1 | blank | p2

# Multi-row with spacers
(p1 + plot_spacer() + p2) /
(p3 + p4 + plot_spacer())

Adding title & annotations

R plot_annotation() 
# Add overall title and subtitle
(p1 + p2) / (p3 + p4) +
  plot_annotation(
    title    = "Motor Trend Car Road Tests",
    subtitle = "Comparison of different plot types",
    caption  = "Data: mtcars dataset"
  )

# Add automatic panel labels (A, B, C, D...)
(p1 + p2) / (p3 + p4) +
  plot_annotation(tag_levels = "A")   # A, B, C, D

# Or numeric labels
(p1 + p2) / (p3 + p4) +
  plot_annotation(tag_levels = "1")   # 1, 2, 3, 4

# Apply same theme to all plots at once
(p1 + p2 + p3 + p4) &
  theme_minimal()   # & applies to all panels

gridExtra

gridExtra's grid.arrange() function is a classic approach to combining plots. It supports custom labels and layout matrices.

R grid.arrange() examples 
install.packages("gridExtra")
library(gridExtra)

# Basic: all 4 plots in automatic layout
grid.arrange(p2, p3, p1, p4)

# Control rows, columns, sizes
grid.arrange(p4, p3, p2, p1,
             top     = "Top label",
             bottom  = "Bottom label",
             left    = "Left label",
             right   = "Right label",
             widths  = c(1, 2),
             heights = c(2, 3))

# Custom layout matrix
# Matrix values are plot numbers (1=p2, 2=p3, 3=p1, 4=p4)
layout <- matrix(c(1, 1,
                    2, 3,
                    4, 4), ncol = 2, byrow = TRUE)

grid.arrange(p2, p3, p1, p4,
             layout_matrix = layout)

# Add a table alongside plots
library(gridExtra)
my_table <- tableGrob(head(mtcars[, 1:4]))
grid.arrange(p1, my_table, ncol = 2)

cowplot

cowplot was developed for publication-quality figures, especially in biology. Its plot_grid() provides fine-grained control over alignment.

R cowplot::plot_grid() 
install.packages("cowplot")
library(cowplot)

# Basic usage
plot_grid(p1, p2, p3, p4)

# With labels and layout control
plot_grid(p1, p2, p3, p4,
          labels  = c("A", "B", "C", "D"),
          ncol    = 2,
          nrow    = 2,
          rel_widths  = c(1, 0.5),
          rel_heights = c(1, 0.5))

# Align axes across plots
plot_grid(p3, p4,
          align  = "hv",   # h=horizontal, v=vertical
          axis   = "tblr") # align all 4 sides

# Nested layouts
top_row <- plot_grid(p1, p2, ncol = 2)
plot_grid(top_row, p3, ncol = 1)
