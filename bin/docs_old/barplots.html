Module 05 — Chart Types

Bar Charts — 5 Examples

A complete guide to drawing barplots in R: from basic Base R charts to advanced grouped and stacked ggplot2 visualizations.

Example Data Setup

All examples in this module use two data frames. Create them at the start of your R session.

R Create example data 
# df1 — Simple data with 4 groups (values + group)
df1 <- data.frame(
  values = c(5, 2, 7, 2),
  group  = LETTERS[1:4]     # "A", "B", "C", "D"
)
df1
##   values group
## 1      5     A
## 2      2     B
## 3      7     C
## 4      2     D

# df2 — Data with main groups AND subgroups
df2 <- data.frame(
  values    = c(3, 2, 6, 3, 4, 3, 8, 2),
  maingroup = rep(LETTERS[1:4], each = 2),
  subgroup  = letters[1:2]  # "a", "b" repeated
)

Example 1 — Base R Barplot

R's built-in barplot() function requires no packages. It's quick for exploratory work but has fewer customization options than ggplot2.

R Default vertical barplot 
# Default barplot — vertical gray bars
barplot(values ~ group, df1)
R Horizontal barplot 
# Use horiz = TRUE for horizontal bars
barplot(values ~ group,
        df1,
        horiz = TRUE)
R Barplot with colors 
# col takes a vector of colors (1 per bar)
barplot(values ~ group,
        df1,
        col = 1:4)    # Colors 1–4 = black, red, green, blue

# Named colors also work
barplot(values ~ group,
        df1,
        col = c("tomato", "steelblue", "darkgreen", "gold"))
R Adding a legend 
barplot(values ~ group, df1, col = 1:4)

# Add legend separately
legend("topright",
       df1$group,
       fill = 1:4)

Example 2 — Basic ggplot2 Bar Chart

ggplot2 uses geom_bar() for bar charts. With pre-summarized data, use stat = "identity" to use the values as-is.

R Default ggplot2 bar chart 
library(ggplot2)

# stat="identity" uses y values directly
ggplot(df1, aes(x = group, y = values)) +
  geom_bar(stat = "identity")
💡
stat options: Use stat = "identity" when y already contains the bar heights. Use stat = "count" (the default) to count rows automatically.
R Horizontal with coord_flip() 
ggplot(df1, aes(x = group, y = values)) +
  geom_bar(stat = "identity") +
  coord_flip()    # Flip x and y axes

# Modern alternative: swap x and y directly
ggplot(df1, aes(x = values, y = group)) +
  geom_bar(stat = "identity")

# Sort bars by value
ggplot(df1, aes(x = values, y = reorder(group, values))) +
  geom_bar(stat = "identity") +
  labs(y = "Group")
R Colored bars with legend 
# Map fill to group — auto-creates legend
ggplot(df1, aes(x = group, y = values, fill = group)) +
  geom_bar(stat = "identity")

# Custom colors
ggplot(df1, aes(x = group, y = values, fill = group)) +
  geom_bar(stat = "identity") +
  scale_fill_manual(values = c("#e84855","#2d6be4","#12a36a","#d4a017")) +
  theme_minimal()

# Add value labels on top
ggplot(df1, aes(x = group, y = values, fill = group)) +
  geom_bar(stat = "identity") +
  geom_text(aes(label = values), vjust = -0.5, size = 4) +
  theme_minimal()

Example 3 — Stacked Bar Chart

Stacked bars show how subgroups contribute to a total. Map fill to the subgroup variable.

R Stacked barplot (ggplot2) 
ggplot(df2, aes(x = maingroup,
             y = values,
             fill = subgroup)) +
  geom_bar(stat = "identity")
# Default: stacked. subgroups stacked within each maingroup bar

Example 4 — 100% Stacked (Proportional)

Scale all bars to the same height to emphasize proportions between subgroups rather than totals.

R Scale to 1.0 
ggplot(df2, aes(x = maingroup, y = values, fill = subgroup)) +
  geom_bar(stat = "identity",
           position = "fill")   # position="fill" scales to 1.0
R Scale to 100% with percent labels 
ggplot(df2, aes(x = maingroup, y = values, fill = subgroup)) +
  geom_bar(stat = "identity", position = "fill") +
  scale_y_continuous(labels = scales::percent_format())
# Y-axis shows 0%, 25%, 50%, 75%, 100%

Example 5 — Grouped Bar Chart

Grouped bars place subgroups side-by-side within each main group. Use position = "dodge".

R Grouped (dodged) barplot 
ggplot(df2, aes(x = maingroup, y = values, fill = subgroup)) +
  geom_bar(stat = "identity",
           position = "dodge")   # "dodge" = side by side

# Control spacing between dodged bars
ggplot(df2, aes(x = maingroup, y = values, fill = subgroup)) +
  geom_bar(stat = "identity",
           position = position_dodge(width = 0.8),
           width = 0.7)

# Full styled example
ggplot(df2, aes(x = maingroup, y = values, fill = subgroup)) +
  geom_bar(stat = "identity", position = "dodge", width = 0.7) +
  geom_text(aes(label = values),
            position = position_dodge(width = 0.7),
            vjust = -0.5, size = 3.5) +
  scale_fill_manual(values = c("#f87060", "#0cc0df")) +
  labs(title = "Grouped Bar Chart",
       x     = "Main Group",
       y     = "Values",
       fill  = "Subgroup") +
  theme_minimal()

position Argument — Quick Comparison

position =EffectWhen to use
"stack"Stacks subgroups on top (default)Show totals + composition
"fill"Stacks to same height (1.0 or 100%)Compare proportions, not totals
"dodge"Places bars side by sideDirect comparison of subgroups
"identity"Overlapping bars (no adjustment)Transparency overlays (rare)
